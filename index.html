const SPREADSHEET_ID = 'YOUR_SPREADSHEET_ID';
const DUMP_TAB_NAME = 'Demo Manager : Petty Cash Voucher wise Dump';
const HO_TAB_1_10 = 'HO Petty Cash - 1st to 10th Nov';
const HO_TAB_11_20 = 'HO Petty Cash - 11th to 20th Nov';
const HO_TAB_21_30 = 'HO Petty Cash - 21st to 30th Nov';
const PHOTO_FOLDER_ID = 'YOUR_DRIVE_FOLDER_ID';
const START_ROW = 8;

function doGet(e) {
  const action = (e && e.parameter && e.parameter.action) ? e.parameter.action : 'getEntries';
  if (action === 'getEntries' || !e || !e.parameter || !e.parameter.ui) {
    try {
      const entries = readAllEntriesFromSheet_();
      return jsonResponse_({ status: 'success', count: entries.length, entries: entries });
    } catch (err) {
      return jsonResponse_({ status: 'error', message: err.toString() });
    }
  }
  return HtmlService.createHtmlOutputFromFile('Index')
    .setTitle('Demo Retail Demo Store (DEMO01) • Petty Cash Web App')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

function doPost(e) {
  const lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try {
    const payload = JSON.parse((e && e.postData && e.postData.contents) ? e.postData.contents : '{}');
    const action = payload.action || 'addEntry';
    let entries = readAllEntriesFromSheet_();
    let savedThumbnailUrl = '';
    let savedFileId = '';

    if (action === 'addEntry') {
      const entry = payload.entry || payload;
      const nextId = entries.reduce((max, item) => Math.max(max, Number(item.id) || 0), 0) + 1;
      let imgUrl = entry.imgData || '';
      if (imgUrl && imgUrl.indexOf('data:image/') === 0) {
        const up = saveBase64ReceiptToDrive_(imgUrl, 'DEMO01-Nov_' + String(entries.length + 1).padStart(2, '0'), entry.staff || 'Staff', entry.date || '2026-11-11');
        imgUrl = up.thumbnailUrl;
        savedThumbnailUrl = up.thumbnailUrl;
        savedFileId = up.fileId;
      }
      entries.push({
        id: entry.id || nextId,
        staff: entry.staff || 'Demo Staff 1',
        paidBy: entry.paidBy || 'Demo Manager',
        date: entry.date || '2026-11-11',
        category: entry.category || 'Travelling',
        remarks: entry.remarks || 'Travelling',
        received: Number(entry.received || 0),
        expense: Number(entry.expense || 0),
        billAmt: Number(entry.billAmt !== undefined ? entry.billAmt : (entry.expense || 0)),
        billNote: entry.billNote || 'Cash Memo',
        pendingRemark: Boolean(entry.pendingRemark),
        imgData: imgUrl
      });
    } else if (action === 'updateEntry') {
      const updated = payload.entry || {};
      let imgUrl = updated.imgData || '';
      if (imgUrl && imgUrl.indexOf('data:image/') === 0) {
        const up = saveBase64ReceiptToDrive_(imgUrl, updated.voucherNo || 'DEMO01-Nov_01', updated.staff || 'Staff', updated.date || '2026-11-01');
        imgUrl = up.thumbnailUrl;
        savedThumbnailUrl = up.thumbnailUrl;
        savedFileId = up.fileId;
      }
      entries = entries.map(item => Number(item.id) === Number(updated.id) ? {
        ...item,
        ...updated,
        expense: Number(updated.expense !== undefined ? updated.expense : item.expense),
        billAmt: Number(updated.billAmt !== undefined ? updated.billAmt : item.billAmt),
        pendingRemark: Boolean(updated.pendingRemark),
        imgData: imgUrl || item.imgData || ''
      } : item);
    } else if (action === 'deleteEntry') {
      entries = entries.filter(item => Number(item.id) !== Number(payload.id));
    } else if (action === 'syncAll' && Array.isArray(payload.entries)) {
      entries = payload.entries.map((item, idx) => {
        let imgUrl = item.imgData || '';
        if (imgUrl && imgUrl.indexOf('data:image/') === 0) {
          imgUrl = saveBase64ReceiptToDrive_(imgUrl, 'DEMO01-Nov_' + String(idx + 1).padStart(2, '0'), item.staff || 'Staff', item.date || '2026-11-01').thumbnailUrl;
        }
        return { ...item, id: Number(item.id) || (idx + 1), imgData: imgUrl };
      });
    }

    const enriched = writeAllEntriesToSheetAndHoTabs_(entries);
    return jsonResponse_({ status: 'success', action: action, thumbnailUrl: savedThumbnailUrl, fileId: savedFileId, entries: enriched });
  } catch (err) {
    return jsonResponse_({ status: 'error', message: err.toString() });
  } finally {
    lock.releaseLock();
  }
}

function saveBase64ReceiptToDrive_(base64DataUrl, voucherNo, staff, dateStr) {
  const parts = base64DataUrl.split(',');
  if (parts.length < 2) return { fileId: '', thumbnailUrl: base64DataUrl };
  const decoded = Utilities.base64Decode(parts[1]);
  const blob = Utilities.newBlob(decoded, 'image/jpeg', voucherNo + '_' + staff + '_' + dateStr + '.jpg');
  const file = DriveApp.getFolderById(PHOTO_FOLDER_ID).createFile(blob);
  return { fileId: file.getId(), thumbnailUrl: 'https://drive.google.com/thumbnail?id=' + file.getId() + '&sz=w1000' };
}

function readAllEntriesFromSheet_() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = ss.getSheetByName(DUMP_TAB_NAME);
  if (!sheet || sheet.getLastRow() < START_ROW) return [];
  const values = sheet.getRange(START_ROW, 1, sheet.getLastRow() - START_ROW + 1, 14).getValues();
  const entries = [];
  for (let i = 0; i < values.length; i++) {
    const r = values[i];
    if (!r[0] && !r[5] && !r[9]) continue;
    const exp = Number(r[9] || 0);
    entries.push({
      id: i + 1,
      staff: String(r[0] || 'Demo Staff 1').trim(),
      paidBy: String(r[1] || 'Demo Manager').trim(),
      date: normalizeToIsoDate_(r[2]),
      voucherNo: 'DEMO01-Nov_' + String(entries.length + 1).padStart(2, '0'),
      category: String(r[4] || 'Travelling').trim(),
      remarks: String(r[5] || '').trim(),
      received: Number(r[8] || 0),
      expense: exp,
      billNote: String(r[11] || 'Cash Memo').trim(),
      billAmt: r[12] !== '' && r[12] !== null ? Number(r[12]) : exp,
      pendingRemark: String(r[5] || '').indexOf('Pending Remark') !== -1,
      imgData: String(r[13] || '').trim()
    });
  }
  return entries;
}

function writeAllEntriesToSheetAndHoTabs_(entries) {
  entries.sort((a, b) => String(a.date).localeCompare(String(b.date)) || ((a.id || 0) - (b.id || 0)));
  let running = 0;
  const enriched = entries.map((item, idx) => {
    const voucherNo = 'DEMO01-Nov_' + String(idx + 1).padStart(2, '0');
    const opening = running;
    const rec = Number(item.received) || 0;
    const exp = Number(item.expense) || 0;
    const closing = opening + rec - exp;
    running = closing;
    const day = parseInt(String(item.date || '2026-11-01').split('-')[2], 10) || 1;
    const bucket = day <= 10 ? 1 : day <= 20 ? 2 : 3;
    const targetSheet = bucket === 1 ? HO_TAB_1_10 : bucket === 2 ? HO_TAB_11_20 : HO_TAB_21_30;
    return {
      ...item,
      id: idx + 1,
      voucherNo,
      opening,
      received: rec,
      expense: exp,
      closing,
      bucket,
      targetSheet,
      billAmt: item.billAmt !== undefined ? Number(item.billAmt) : exp,
      pendingRemark: String(item.remarks || '').indexOf('Pending Remark') !== -1
    };
  });

  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const dumpSheet = ss.getSheetByName(DUMP_TAB_NAME);
  if (dumpSheet) {
    const maxClear = Math.max(dumpSheet.getLastRow(), START_ROW + enriched.length + 5);
    dumpSheet.getRange(START_ROW, 1, maxClear - START_ROW + 1, 14).clearContent();
    if (enriched.length > 0) {
      dumpSheet.getRange(START_ROW, 1, enriched.length, 14).setValues(enriched.map(r => [
        r.staff, r.paidBy, formatDisplayDate_(r.date), r.voucherNo, r.category, r.remarks,
        r.targetSheet, r.opening, r.received, r.expense, r.closing, r.billNote, r.billAmt, r.imgData || ''
      ]));
    }
  }
  syncHoPeriodSheet_(ss, HO_TAB_1_10, enriched.filter(x => x.bucket === 1 && x.expense > 0));
  syncHoPeriodSheet_(ss, HO_TAB_11_20, enriched.filter(x => x.bucket === 2 && x.expense > 0));
  syncHoPeriodSheet_(ss, HO_TAB_21_30, enriched.filter(x => x.bucket === 3 && x.expense > 0));
  return enriched;
}

function syncHoPeriodSheet_(ss, tabName, rows) {
  const sheet = ss.getSheetByName(tabName);
  if (!sheet) return;
  sheet.getRange(12, 2, 60, 5).clearContent();
  if (rows.length > 0) {
    sheet.getRange(12, 2, Math.min(rows.length, 60), 5).setValues(rows.slice(0, 60).map(r => [
      formatDisplayDate_(r.date), r.remarks, r.category, r.voucherNo, r.expense
    ]));
  }
}

function normalizeToIsoDate_(val) {
  if (!val) return '2026-11-01';
  if (Object.prototype.toString.call(val) === '[object Date]' && !isNaN(val.getTime())) {
    return Utilities.formatDate(val, 'Asia/Kolkata', 'yyyy-MM-dd');
  }
  const s = String(val).trim();
  if (/^\d{4}-\d{2}-\d{2}$/.test(s)) return s;
  const d = new Date(s);
  return !isNaN(d.getTime()) ? Utilities.formatDate(d, 'Asia/Kolkata', 'yyyy-MM-dd') : '2026-11-01';
}

function formatDisplayDate_(iso) {
  const m = ['Jan','Feb','Mar','Apr','May','Jun','Jul','Aug','Sep','Oct','Nov','Dec'];
  const p = String(iso || '2026-11-01').split('-');
  return p.length === 3 ? (p[2] + ' ' + (m[(parseInt(p[1], 10) || 11) - 1] || 'Nov') + ' ' + p[0].slice(-2)) : iso;
}

function jsonResponse_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
